cmake_minimum_required(VERSION 3.24)
project(WindowsWhip VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(WHIP_WITH_AUDIO "Embed the locally prepared sound pack" ON)
option(WHIP_ASAN "Enable AddressSanitizer for development" OFF)
add_library(whip_core STATIC src/core.cpp)
target_include_directories(whip_core PUBLIC src)
if(MSVC)
  target_compile_options(whip_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
  if(WHIP_ASAN)
    target_compile_options(whip_core PUBLIC /fsanitize=address /Zi)
    target_link_options(whip_core PUBLIC /INCREMENTAL:NO)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(whip_tests tests/core_tests.cpp)
  target_link_libraries(whip_tests PRIVATE whip_core)
  add_test(NAME core COMMAND whip_tests)
endif()
if(WIN32)
  enable_language(RC)
  set(WHIP_AUDIO_RC "")
  if(WHIP_WITH_AUDIO)
    foreach(sample IN ITEMS hit_01 swing_01 swing_02)
      if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resources/audio/${sample}.wav")
        message(FATAL_ERROR "Run python tools/prepare_audio.py first, or use -DWHIP_WITH_AUDIO=OFF for CI.")
      endif()
    endforeach()
    set(WHIP_AUDIO_RC "101 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/resources/audio/hit_01.wav\"\n102 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/resources/audio/swing_01.wav\"\n103 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/resources/audio/swing_02.wav\"")
  endif()
  configure_file(resources/whip.rc.in "${CMAKE_CURRENT_BINARY_DIR}/whip.rc" @ONLY)
  set(resource_inputs "${CMAKE_CURRENT_SOURCE_DIR}/resources/whip.manifest" "${CMAKE_CURRENT_SOURCE_DIR}/resources/whip.ico" "${CMAKE_CURRENT_SOURCE_DIR}/resources/manual_en.txt" "${CMAKE_CURRENT_SOURCE_DIR}/resources/manual_ru.txt")
  if(WHIP_WITH_AUDIO)
    foreach(sample IN ITEMS hit_01 swing_01 swing_02)
      list(APPEND resource_inputs "${CMAKE_CURRENT_SOURCE_DIR}/resources/audio/${sample}.wav")
    endforeach()
  endif()
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/whip.rc" PROPERTIES OBJECT_DEPENDS "${resource_inputs}")
  add_executable(Whip WIN32 src/main.cpp src/flyout.cpp src/platform.cpp src/renderer.cpp src/audio.cpp "${CMAKE_CURRENT_BINARY_DIR}/whip.rc")
  target_link_libraries(Whip PRIVATE whip_core d2d1 dwrite d3d11 dxgi dcomp dwmapi magnification xaudio2 shell32 user32 gdi32 ole32 wtsapi32 winmm comctl32)
  target_compile_definitions(Whip PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_options(Whip PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
  if(BUILD_TESTING)
    add_executable(whip_preview WIN32 tests/preview.cpp)
    target_compile_definitions(whip_preview PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(whip_preview PRIVATE shell32)
    add_executable(whip_input_test WIN32 tests/input_test.cpp src/platform.cpp)
    target_compile_definitions(whip_input_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(whip_input_test PRIVATE whip_core user32 shell32 ole32)
    add_executable(platform_tests tests/platform_tests.cpp src/platform.cpp)
    target_compile_definitions(platform_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(platform_tests PRIVATE whip_core user32 shell32 ole32)
    add_test(NAME platform COMMAND platform_tests)
    add_executable(ui_tests tests/ui_tests.cpp src/flyout.cpp)
    target_compile_definitions(ui_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(ui_tests PRIVATE whip_core d2d1 dwrite user32)
    configure_file(resources/manual_en.txt test_manual_en.txt COPYONLY)
    configure_file(resources/manual_ru.txt test_manual_ru.txt COPYONLY)
    add_test(NAME ui_layout COMMAND ui_tests)
    add_test(NAME application_smoke COMMAND Whip --smoke-test)
    add_test(NAME pause_animation COMMAND Whip --animation-test)
    set_tests_properties(application_smoke PROPERTIES TIMEOUT 15)
    set_tests_properties(pause_animation PROPERTIES TIMEOUT 15)
  endif()
  install(TARGETS Whip RUNTIME DESTINATION .)
endif()
